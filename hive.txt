# JOIN QUERY

```sql id="5m1q8p"
SELECT

c.cust_name,

o.order_id,

i.item_name,

o.quantity,

i.item_price

FROM customer_info c

JOIN order_info o

ON c.order_id = o.order_id

JOIN item_info i

ON o.item_id = i.item_id;
```

---

# TOTAL SALES

```sql id="9v4m2x"
SELECT
SUM(o.quantity * i.item_price)

FROM order_info o

JOIN item_info i

ON o.item_id = i.item_id;
```

---

# AVERAGE SALES

```sql id="3n7q1v"
SELECT
AVG(o.quantity * i.item_price)

FROM order_info o

JOIN item_info i

ON o.item_id = i.item_id;
```

---

# MAXIMUM COST ORDER

```sql id="6x2m5q"
SELECT

o.order_id,

i.item_name,

(o.quantity * i.item_price) AS total_cost

FROM order_info o

JOIN item_info i

ON o.item_id = i.item_id

ORDER BY total_cost DESC

LIMIT 1;
```

# EXTERNAL HIVE TABLE CONNECTED TO HBASE

# =========================================

```sql id="3v8n1k"
CREATE EXTERNAL TABLE hive_table(

key STRING,

name STRING,

value STRING

)

STORED BY
'org.apache.hadoop.hive.hbase.HBaseStorageHandler'

WITH SERDEPROPERTIES(

"hbase.columns.mapping"=
":key,info:name,info:value"

)

TBLPROPERTIES(

"hbase.table.name"="table_name"

);
```
